namespace q
{
    class Program
    {
        static void Main()
        {
            int[] numeros = { 10, 20, 30, 40, 50, 0 };

            Console.Write("Ingrese un nuevo número: ");
            int nuevoNumero = int.Parse(Console.ReadLine());

            for (int i = numeros.Length - 1; i > 2; i--)
            {
                numeros[i] = numeros[i - 1];
            }

            numeros[2] = nuevoNumero;

            Console.WriteLine("Arreglo resultante:");

            foreach (int numero in numeros)
            {
                Console.Write(numero + " ");
            }
        }
    }
}
