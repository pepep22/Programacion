namespace q
{
    class Program
    {
        static void Main()
        {
            string[] ciudades = { "Madrid", "Buenos Aires", "Bogotá", "Lima", "Santiago" };

            Console.Write("Ingrese el nombre de una ciudad: ");
            string ciudadBuscada = Console.ReadLine();

            int posicion = -1;

            for (int i = 0; i < ciudades.Length; i++)
            {
                if (ciudades[i].Equals(ciudadBuscada, StringComparison.OrdinalIgnoreCase))
                {
                    posicion = i;
                    break;
                }
            }

            if (posicion != -1)
            {
                Console.WriteLine($"Ciudad encontrada en la posición (índice): {posicion}");
            }
            else
            {
                Console.WriteLine("Ciudad no encontrada");
            }
        }
    }
}
