namespace q
{
    class Program
    {
        static void Main()
        {
            int mayor = int.MinValue;

            for (int i = 1; i <= 10; i++)
            {
                Console.Write("Número " + i + ": ");
                int num = int.Parse(Console.ReadLine());

                if (num > mayor)
                {
                    mayor = num;
                }
            }

            Console.WriteLine("El número más alto fue: " + mayor);
        }
    }
}
