namespace q
{
    class Program
    {
        static void Main()
        {
            Console.Write("Ingresá tu edad: ");
            int edad = int.Parse(Console.ReadLine());

            if (edad < 18)
            {
                Console.WriteLine("Sos menor de edad.");
            }
            else if (edad == 18)
            {
                Console.WriteLine("Tenés justo 18 años.");
            }
            else
            {
                Console.WriteLine("Sos mayor de edad.");
            }
        }
    }
}
