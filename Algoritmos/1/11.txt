namespace q
{
    class Program
    {
        static void Main()
        {
            Console.Write("Ingresá un número: ");
            int num = int.Parse(Console.ReadLine());

            for (int i = 1; i <= 12; i++)
            {
                Console.WriteLine($"{num} x {i} = {num * i}");
            }
        }
    }
}
