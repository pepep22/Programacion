namespace q
{
    class Program
    {
        static void Main()
        {
            double total = 0;
            int productos = 0;

            while (total <= 1000)
            {
                Console.Write("Precio: ");
                total += double.Parse(Console.ReadLine());
                productos++;
            }

            Console.WriteLine($"Productos cargados: {productos}");
        }
    }
}
