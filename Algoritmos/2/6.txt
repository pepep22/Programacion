namespace q
{
    class Program
    {
        static void Main()
        {
            List<string> turnos = new List<string>();

            for (int i = 1; i <= 5; i++)
            {
                Console.Write("Nombre: ");
                turnos.Add(Console.ReadLine());
            }

            for (int i = 0; i < turnos.Count; i++)
            {
                Console.WriteLine($"Turno {i + 1}: {turnos[i]}");
            }
        }
    }
}
