namespace q
{
    class Program
    {
        static void Main()
        {
            Random rnd = new Random();

            string[] opciones = { "piedra", "papel", "tijera" };

            int usuario = 0;
            int pc = 0;

            for (int ronda = 1; ronda <= 3; ronda++)
            {
                Console.Write("Elegí piedra, papel o tijera: ");
                string jugada = Console.ReadLine().ToLower();

                string maquina = opciones[rnd.Next(3)];

                Console.WriteLine($"Computadora: {maquina}");

                if (
                    (jugada == "piedra" && maquina == "tijera") ||
                    (jugada == "papel" && maquina == "piedra") ||
                    (jugada == "tijera" && maquina == "papel")
                )
                    usuario++;
                else if (jugada != maquina)
                    pc++;
            }

            Console.WriteLine($"Usuario: {usuario}");
            Console.WriteLine($"Computadora: {pc}");
        }
    }
}
