namespace q
{
    class Program
    {
        static void Main()
        {
            string respuesta;

            do
            {
                Console.Write("Celsius: ");
                double c = double.Parse(Console.ReadLine());

                double f = c * 9 / 5 + 32;

                Console.WriteLine($"Fahrenheit: {f}");

                Console.Write("¿Continuar? (s/n): ");
                respuesta = Console.ReadLine().ToLower();

            } while (respuesta == "s");
        }
    }
}
