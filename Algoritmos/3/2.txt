namespace q
{
    class Program
    {
        static void Main()
        {
            Console.Write("Número: ");
            int num = int.Parse(Console.ReadLine());

            if (num < 1)
            {
                Console.WriteLine("No se puede.");
            }
            else
            {
                for (int i = 1; i <= num; i++)
                {
                    if (i % 2 != 0)
                        Console.WriteLine(i);
                }
            }
        }
    }
}
