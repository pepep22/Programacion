namespace q
{
    class Program
    {
        static void Main()
        {
            Random rnd = new Random();
            int secreto = rnd.Next(1, 101);

            int intentos = 0;

            while (true)
            {
                Console.Write("Adiviná: ");
                int numero = int.Parse(Console.ReadLine());

                intentos++;

                if (numero < secreto)
                    Console.WriteLine("Más grande");
                else if (numero > secreto)
                    Console.WriteLine("Más chico");
                else
                {
                    Console.WriteLine($"¡Correcto! Intentos: {intentos}");
                    break;
                }
            }
        }
    }
}
